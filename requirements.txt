# Runtime libraries for the Printer (the Pi and the Mac). Dev and test tools are in requirements-dev.txt.
# Install into the project venv:  python -m pip install -r requirements.txt
# Both are pure Python (their own dependencies ship wheels for aarch64); verify on the Pi when deploying.

# The monitoring site: a web page of the last hour of turns (pom.monitor). Pure Python.
flask>=3.0,<4

# Everyday-speech word frequencies, for "how rare is this word in normal talk".
wordfreq>=3.1,<4

# WordNet (senses and relations), a part-of-speech tagger and stopword lists, for reading the
# meaning of a fragment. Its data is fetched separately: scripts/setup/setup-nltk-data.sh
nltk>=3.9,<4

# The serial port to the typewriter (pom.printing.TerminalPort). Pure Python, the standard choice.
pyserial>=3.5,<4

# The machine's network addresses, for the boot slip (pom.conductor.AddressFinder). Prebuilt wheels for the Pi (aarch64) and the Mac.
psutil>=5.9,<8
